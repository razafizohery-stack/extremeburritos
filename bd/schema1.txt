-- Extensions nécessaires
CREATE EXTENSION IF NOT EXISTS "pgcrypto";
CREATE EXTENSION IF NOT EXISTS "uuid-ossp";

-- =========================
-- admin_settings
-- =========================
CREATE TABLE admin_settings (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    key VARCHAR(50) NOT NULL,
    value VARCHAR(255) NOT NULL
);

-- =========================
-- categories
-- =========================
CREATE TABLE categories (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL,
    name TEXT NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc', now())
);

-- =========================
-- clients
-- =========================
CREATE TABLE clients (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID,
    name TEXT NOT NULL,
    email TEXT,
    phone TEXT,
    address TEXT,
    created_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc', now()),
    nif TEXT,
    stat TEXT
);

-- =========================
-- daily_closures
-- =========================
CREATE TABLE daily_closures (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    closure_date DATE NOT NULL,
    total_cash NUMERIC DEFAULT 0,
    total_credit NUMERIC DEFAULT 0,
    total_advances NUMERIC DEFAULT 0,
    total_journalier NUMERIC DEFAULT 0,
    user_id UUID,
    created_at TIMESTAMPTZ DEFAULT timezone('utc', now())
);

-- =========================
-- decaissements
-- =========================
CREATE TABLE decaissements (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID,
    motif TEXT NOT NULL,
    montant NUMERIC NOT NULL DEFAULT 0,
    categorie TEXT,
    date_decaissement DATE DEFAULT CURRENT_DATE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc', now())
);

-- =========================
-- delivery_notes
-- =========================
CREATE TABLE delivery_notes (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    supplier_id UUID,
    bl_number VARCHAR(50),
    bl_date TIMESTAMPTZ NOT NULL DEFAULT timezone('utc', now()),
    total_amount NUMERIC DEFAULT 0.00,
    payment_type VARCHAR(20) DEFAULT 'direct_sale',
    due_date DATE,
    advance_amount NUMERIC DEFAULT 0.00,
    user_id UUID,
    created_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc', now()),
    credit_frequency VARCHAR(40),
    supplier_invoice_number VARCHAR(50),
    destination_depot VARCHAR(100) DEFAULT 'Magasin Principal',
    credit_type TEXT,
    first_due_date DATE,
    total_initial NUMERIC DEFAULT 0.00,
    type VARCHAR(20) DEFAULT 'in'
);

-- =========================
-- delivery_note_items
-- =========================
CREATE TABLE delivery_note_items (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    delivery_note_id UUID NOT NULL,
    product_id UUID,
    quantity INTEGER NOT NULL,
    purchase_price_per_unit NUMERIC NOT NULL,
    selling_price_per_unit NUMERIC,
    line_total_purchase NUMERIC NOT NULL,
    line_total_selling NUMERIC,
    unit VARCHAR(50) DEFAULT 'base',
    superior_unit_name VARCHAR(50),
    created_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc', now()),

    CONSTRAINT fk_delivery_note
        FOREIGN KEY (delivery_note_id)
        REFERENCES delivery_notes(id)
        ON DELETE CASCADE
);

-- =========================
-- depots
-- =========================
CREATE TABLE depots (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    name VARCHAR(100) NOT NULL,
    location VARCHAR(255),
    created_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc', now()),
    phone VARCHAR(50),
    email VARCHAR(100),
    nif VARCHAR(50),
    stat VARCHAR(50),
    address TEXT
);

-- =========================
-- factures
-- =========================
CREATE TABLE factures (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID NOT NULL,
    client_id UUID,
    number TEXT NOT NULL,
    total_amount NUMERIC NOT NULL DEFAULT 0,
    status TEXT DEFAULT 'draft',
    due_date DATE,
    created_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc', now()),
    guest_name TEXT,
    guest_contact TEXT,
    guest_nif TEXT,
    guest_stat TEXT,
    paid_amount NUMERIC DEFAULT 0,
    month SMALLINT,
    year INTEGER,

    CONSTRAINT fk_facture_client
        FOREIGN KEY (client_id)
        REFERENCES clients(id)
        ON DELETE SET NULL
);

-- =========================
-- facture_items
-- =========================
CREATE TABLE facture_items (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    facture_id UUID NOT NULL,
    produit_id UUID,
    quantity INTEGER NOT NULL,
    unit_price NUMERIC NOT NULL,
    created_at TIMESTAMPTZ NOT NULL DEFAULT timezone('utc', now()),
    discount_value NUMERIC DEFAULT 0,
    discount_type TEXT DEFAULT 'percent',
    discount JSONB,
    unit_type VARCHAR(20) DEFAULT 'base',
    total DOUBLE PRECISION,

    CONSTRAINT fk_facture
        FOREIGN KEY (facture_id)
        REFERENCES factures(id)
        ON DELETE CASCADE
);

-- =========================
-- echeances_details
-- =========================
CREATE TABLE echeances_details (
    id UUID PRIMARY KEY DEFAULT uuid_generate_v4(),
    facture_id UUID,
    montant NUMERIC NOT NULL,
    date_echeance DATE NOT NULL,
    statut TEXT DEFAULT 'non_paye',
    created_at TIMESTAMPTZ DEFAULT now(),

    CONSTRAINT fk_echeance_facture
        FOREIGN KEY (facture_id)
        REFERENCES factures(id)
        ON DELETE CASCADE
);
